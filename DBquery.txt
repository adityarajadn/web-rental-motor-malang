-- Enable UUID Extension
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- =====================================================
-- USERS
-- =====================================================

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(150) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    phone VARCHAR(20),
    role VARCHAR(20) NOT NULL DEFAULT 'customer'
        CHECK (role IN ('admin','customer')),
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP
);

-- =====================================================
-- MOTORS
-- =====================================================

CREATE TABLE motors (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name VARCHAR(150) NOT NULL,
    brand VARCHAR(100) NOT NULL,
    type VARCHAR(20) NOT NULL
        CHECK (type IN ('Matic','Manual','Sport')),

    year INT NOT NULL,
    cc INT NOT NULL,

    transmission VARCHAR(20) NOT NULL
        CHECK (transmission IN ('Automatic','Manual')),

    license_plate VARCHAR(20) UNIQUE NOT NULL,

    price_per_day NUMERIC(12,2) NOT NULL,

    image_url TEXT NOT NULL,

    status VARCHAR(20) NOT NULL DEFAULT 'available'
        CHECK (status IN ('available','rented','maintenance')),

    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP
);

-- =====================================================
-- BOOKINGS
-- =====================================================

CREATE TABLE bookings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    booking_code VARCHAR(20) UNIQUE NOT NULL,

    user_id UUID NOT NULL,
    motor_id UUID NOT NULL,

    start_date TIMESTAMP NOT NULL,
    end_date TIMESTAMP NOT NULL,

    pickup_location TEXT NOT NULL,
    dropoff_location TEXT NOT NULL,

    total_days INT NOT NULL,
    total_price NUMERIC(12,2) NOT NULL,

    status VARCHAR(30) NOT NULL DEFAULT 'pending_payment'
        CHECK (
            status IN (
                'pending_payment',
                'awaiting_verification',
                'confirmed',
                'active',
                'completed',
                'cancelled'
            )
        ),

    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP,

    CONSTRAINT fk_booking_user
        FOREIGN KEY (user_id)
        REFERENCES users(id),

    CONSTRAINT fk_booking_motor
        FOREIGN KEY (motor_id)
        REFERENCES motors(id)
);

-- =====================================================
-- PAYMENTS
-- =====================================================

CREATE TABLE payments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    booking_id UUID UNIQUE NOT NULL,

    amount NUMERIC(12,2) NOT NULL,

    payment_method VARCHAR(30) NOT NULL
        CHECK (payment_method IN ('Bank Transfer','QRIS')),

    proof_image_url TEXT NOT NULL,

    status VARCHAR(20) NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending','verified','rejected')),

    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP,

    CONSTRAINT fk_payment_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(id)
        ON DELETE CASCADE
);

-- =====================================================
-- HANDOVERS
-- =====================================================

CREATE TABLE handovers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    booking_id UUID NOT NULL,

    type VARCHAR(20) NOT NULL
        CHECK (type IN ('pickup','dropoff')),

    location TEXT NOT NULL,

    scheduled_time TIMESTAMP NOT NULL,
    actual_time TIMESTAMP,

    status VARCHAR(20) NOT NULL DEFAULT 'pending'
        CHECK (status IN ('pending','completed')),

    condition_notes TEXT,

    created_at TIMESTAMP DEFAULT NOW(),

    CONSTRAINT fk_handover_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(id)
        ON DELETE CASCADE
);

-- =====================================================
-- TESTIMONIALS
-- =====================================================

CREATE TABLE testimonials (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id UUID NOT NULL,
    booking_id UUID UNIQUE NOT NULL,

    rating INT NOT NULL
        CHECK (rating BETWEEN 1 AND 5),

    text TEXT NOT NULL,

    is_featured BOOLEAN NOT NULL DEFAULT FALSE,

    created_at TIMESTAMP DEFAULT NOW(),

    CONSTRAINT fk_testimonial_user
        FOREIGN KEY (user_id)
        REFERENCES users(id),

    CONSTRAINT fk_testimonial_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(id)
        ON DELETE CASCADE
);

-- =====================================================
-- SETTINGS
-- =====================================================

CREATE TABLE settings (
    id SERIAL PRIMARY KEY,

    key VARCHAR(100) UNIQUE NOT NULL,

    value TEXT NOT NULL,

    description VARCHAR(255),

    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_users_email
ON users(email);

CREATE INDEX idx_bookings_user
ON bookings(user_id);

CREATE INDEX idx_bookings_motor
ON bookings(motor_id);

CREATE INDEX idx_bookings_status
ON bookings(status);

CREATE INDEX idx_bookings_start_date
ON bookings(start_date);

CREATE INDEX idx_payments_status
ON payments(status);

CREATE INDEX idx_motors_status
ON motors(status);

CREATE INDEX idx_handovers_booking
ON handovers(booking_id);

CREATE INDEX idx_testimonials_featured
ON testimonials(is_featured);
